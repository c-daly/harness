"""Operator host for a plugin-selected native root worker with live time grants.

Uses the existing root timer extension API so the operator can inspect progress
and grant time without cancelling a working session. No overall trial deadline.
"""
import argparse
import asyncio
from dataclasses import asdict, replace
import fcntl
import hashlib
import json
import os
from pathlib import Path
import signal
import sys

ROOT = Path('/home/fearsidhe/projects/harness')
HOST = ROOT / '.worktrees/harness-reliability'
sys.path.insert(0, str(HOST / 'plugins/agent-swarm-runner'))
from run_one import build_native_kernel, command, plugin_command, save, worker_prompt
from run_verified import PinnedRoute
from harness.agent import AgentTask, TaskLimits
from harness.events import CustomEvent
from harness.hooks import Allow, Block, ProposedModelCall
from harness.usage_budget import project_usage
from harness.log import read_session


class RootWorker:
    async def __call__(self, action):
        if not isinstance(action, ProposedModelCall) and str(action.tool) == 'dispatch_agent':
            return Block('this selected workflow task is assigned to this native root worker')
        return Allow()


async def run(args, request):
    workspace = Path(request['worktree_dir']).resolve(strict=True)
    if command(['git', 'branch', '--show-current'], cwd=workspace).strip() != request['branch_name']:
        raise ValueError('plugin branch ownership mismatch')
    if command(['git', 'status', '--porcelain'], cwd=workspace):
        raise ValueError('checkpoint existing work before starting a fresh run')
    kernel, resolved = build_native_kernel(args, workspace, max_children=0, resume=args.root_id)
    kernel.hooks.register_dispatch('pinned-route', PinnedRoute(args.model))
    kernel.hooks.register_dispatch('root-worker', RootWorker())
    from harness.context import ContextPolicy
    from harness.events import ContextPolicyConfigured
    scope = kernel.loop.dispatcher.scope
    policy = ContextPolicy.model_validate(kernel.context_policy.model_dump() | {'max_input_bytes': 512 * 1024})
    event = ContextPolicyConfigured(policy=policy)
    if kernel.session.append(event).event != event:
        raise ValueError('operator context configuration was rewritten')
    updated_scope = replace(scope, context_policy=policy)
    kernel.loop.dispatcher.scope = updated_scope
    kernel.runner._root_scopes[str(kernel.session.id)] = updated_scope
    kernel.loop.max_iterations = 128
    prompt = worker_prompt(request) + "\n\nCURRENT OPERATOR REVIEW AND CONTINUATION INSTRUCTIONS:\n" + args.feedback.read_text()
    prompt += '\n\nThis request is assigned directly to this native Harness root worker. Do not delegate. '
    prompt += (HOST/'plugins/agent-swarm-runner/bundle/agent-swarm-runner/agents/swarm-implementer.md').read_text().split('---',2)[2]
    task = AgentTask(prompt=prompt, limits=TaskLimits(max_iterations=128, max_output_tokens=16384))
    budget = kernel.runner.scope_for(kernel.session).budget
    from harness.execution import ExecutionLimits
    from harness.execution_controls import configure_execution
    normal_tool_limit = ExecutionLimits().max_tool_calls
    if budget.limits.max_tool_calls != normal_tool_limit:
        kernel.session.append(CustomEvent(namespace='operator',name='execution_default_restored',
            data={'setting':'max_tool_calls','previous':budget.limits.max_tool_calls,
                  'value':normal_tool_limit,'reason':'Use the normal core tool capacity for continued development; preserve all cumulative counts.'}))
        configure_execution(kernel, {'max_tool_calls':normal_tool_limit})
    launch = dict(request=request, root_session_id=str(kernel.session.id), pid=os.getpid(),
                  route=str(resolved.route), host_head=command(['git','rev-parse','HEAD'],cwd=HOST).strip(),
                  candidate_head=command(['git','rev-parse','HEAD'],cwd=workspace).strip(),
                  host_sha256=hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
                  task_limits=task.limits.model_dump(), explicit_task_fields=sorted(task.limits.model_fields_set),
                  execution_limits=asdict(budget.limits), overall_deadline=None,
                  context_policy=kernel.context_policy.model_dump(mode='json'),
                  context_capacity_evidence={'installed_provider_max_input_tokens':272000,'last_successful_input_tokens':54325,'operator_max_input_bytes':524288},
                  binding='installed agent-swarm queue; native Harness root worker; operator live grants',
                  queue_completion_recorded=False)
    save(args.output/'launch.json',launch)
    (args.output/'prompt.txt').write_text(prompt)
    assert kernel.resumed
    assert budget.model_calls == args.previous_result["model_calls"] and budget.tool_calls == args.previous_result["tool_calls"], "resume must retain cumulative counts"
    assert json.loads(plugin_command(args,'status'))['tasks'][request['task_name']]['worker_id'] == f'harness:{kernel.session.id}'
    print(json.dumps(launch),flush=True)
    running = asyncio.current_task()
    loop=asyncio.get_running_loop()
    for sig in (signal.SIGINT,signal.SIGTERM):
        loop.add_signal_handler(sig,running.cancel)

    async def monitor():
        while True:
            rows=budget.runs.snapshot(kernel.session.id)
            save(args.output/'live.json',dict(root_session_id=str(kernel.session.id),
                 budgets=[asdict(r) for r in rows], activity=[asdict(r) for r in budget.activity.snapshot()],
                 model_calls=budget.model_calls, tool_calls=budget.tool_calls))
            control=args.output/'control.json'
            if control.exists():
                data=json.loads(control.read_text())
                ack=args.output/('grant-'+data['id']+'.json')
                if not ack.exists():
                    try:
                        kernel.session.append(CustomEvent(namespace='operator',name='progress_review',data=data))
                        status=budget.runs.extend(kernel.session,data['run_id'],data['seconds'])
                        save(ack,dict(status='extended',budget=asdict(status),reason=data['reason']))
                    except Exception as exc:
                        save(ack,dict(status='refused',error=str(exc)))
            await asyncio.sleep(2)

    monitoring=asyncio.create_task(monitor())
    result=None
    error=None
    try:
        result=await kernel.loop.run_task(task)
        (args.output/'worker-result.txt').write_text(result.read_text(kernel.session.blobs))
    except BaseException as exc:
        error=type(exc).__name__+': '+str(exc)
        raise
    finally:
        monitoring.cancel()
        await asyncio.gather(monitoring,return_exceptions=True)
        for sig in (signal.SIGINT,signal.SIGTERM):
            loop.remove_signal_handler(sig)
        try:
            await kernel.loop.end()
            usage=project_usage(read_session(kernel.session.base,kernel.session.id,repair=False))
            save(args.output/'result.json',dict(root_session_id=str(kernel.session.id),
                 result=result.model_dump(mode='json') if result else None,error=error,
                 model_calls=budget.model_calls,tool_calls=budget.tool_calls,
                 usage=dict(input_tokens=usage.input_tokens,output_tokens=usage.output_tokens,
                            cost_usd=str(usage.cost_usd),unknown_cost=usage.unknown_cost),
                 queue_completion_recorded=False,git_status=command(['git','status','--porcelain'],cwd=workspace)))
        finally:
            await kernel.resources.close(emit=kernel.session.append)
            kernel.session.close()


def main():
    p=argparse.ArgumentParser()
    p.add_argument('--output',type=Path,required=True)
    p.add_argument('--previous',type=Path,required=True)
    p.add_argument('--feedback',type=Path,required=True)
    args=p.parse_args()
    args.plugin_root=Path('/home/fearsidhe/.claude/plugins/agent-swarm')
    args.manifest=ROOT/'.worktrees/progress-aware-completion/docs/handoffs/2026-09-13-reliability-integration/progress-plan-manifest.yaml'
    args.state_dir=ROOT/'.worktrees/tmp/progress-continuation-state'
    args.catalog=Path('/home/fearsidhe/.config/harness/models.toml')
    args.model='gpt'
    args.max_model_calls=1024
    args.timeout=600
    args.previous_result=json.loads((args.previous/'result.json').read_text())
    previous_launch=json.loads((args.previous/'launch.json').read_text())
    args.root_id=previous_launch['root_session_id']
    error=args.previous_result.get('error') or ''
    outcome=args.previous_result.get('result') or {}
    if not (error.startswith('ContextOverflow:') or outcome.get('status')=='completed'
            or (outcome.get('status')=='incomplete' and outcome.get('reason')=='iteration_limit')):
        raise ValueError('previous stop requires separate effect/authority reconciliation')
    journal=(args.previous/'journal').resolve(strict=True)
    events=[json.loads(line)['event'] for line in (journal/'sessions'/f'{args.root_id}.jsonl').read_text().splitlines()]
    proposed={e['call_id'] for e in events if e['type']=='tool_call_proposed'}
    settled={e['call_id'] for e in events if e['type']=='tool_call_completed'}
    if proposed != settled or events[-1]['type']!='session_ended':
        raise ValueError('previous native session has unsettled tools or no terminal lifecycle')
    args.output.mkdir(exist_ok=False)
    (args.output/'journal').symlink_to(journal,target_is_directory=True)
    with (args.state_dir/'.harness-runner.lock').open('a') as lock:
        fcntl.flock(lock,fcntl.LOCK_EX|fcntl.LOCK_NB)
        requests=[previous_launch['request']]
        save(args.output/'pending.json',requests)
        (args.output/'queue-before.json').write_text(plugin_command(args,'status'))
        if not requests:
            raise ValueError('no eligible plugin task')
        asyncio.run(run(args,requests[0]))

if __name__=='__main__':
    main()
